% Define an entry of coursework information
% Usage: \centry{<title>}{<location>}{<date>}{<description>}
\newcommand*{\centry}[4]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \entrytitlestyle{#1\ifempty{#2}{}{,~\entrylocationstyle{#2}}} & \multirow[t]{2}{\metacolumnsize}{\raggedleft\entrydatestyle{#3}} \\
    \descriptionstyle{#4} & {}
  \end{tabular*}%
}

% Define an entry of project information
% Usage: \pentry{<title>}{<date>}{<description>}
\newcommand*{\pentry}[3]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \multicolumn{2}{L{\textwidth}}{\entrytitlestyle{#1 (#2)}} \\
    \descriptionstyle{#3} & {}
  \end{tabular*}%
}

% Define an entry of cv information
% Usage: \cventry{<position>}{<title>}{<location>}{<date>}{<description>}
\renewcommand*{\cventry}[5]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \ifempty{#1#3}
      {\entrypositionstyle{#2} & \entrydatestyle{#4} \\}
      {\entrytitlestyle{#1} & \multirow[t]{3}{\metacolumnsize}{\raggedleft\entrydatestyle{#4}} \\
      \entrypositionstyle{#2\ifempty{#3}{}{,~\entrylocationstyle{#3}}} &\\}
    \descriptionstyle{#5} &
  \end{tabular*}%
}
