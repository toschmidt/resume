% Define an entry of coursework information
% Usage: \centry{<title>}{<location>}{<date>}{<description>}
\newcommand*{\centry}[4]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \entrytitlestyle{#1\ifempty{#2}{}{,~\entrylocationstyle{#2}}} & \entrydatestyle{#3} \\
    \multicolumn{2}{L{\textwidth}}{\descriptionstyle{#4}}
  \end{tabular*}%
}

% Define an entry of project information
% Usage: \pentry{<title>}{<date>}{<description>}
\newcommand*{\pentry}[3]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \multicolumn{2}{L{\textwidth}}{\entrytitlestyle{#1 (#2)}} \\
    \descriptionstyle{#3} & {}
  \end{tabular*}%
}

% Define an entry of cv information
% Usage: \cventry{<job title>}{<organization>}{<location>}{<date>}{<description>}
\renewcommand*{\cventry}[5]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - \metacolumnsize} R{\metacolumnsize}}
    \ifempty{#1#3}
      {\entrypositionstyle{#2} & \entrydatestyle{#4} \\}
      {\ifempty{#3}
      {\entrytitlestyle{#1} & \entrylocationstyle{#4} \\
      \entrypositionstyle{#2} & }
      {\entrytitlestyle{#1} & \entrylocationstyle{#3} \\
      \entrypositionstyle{#2} & \entrydatestyle{#4}} \\}
      \multicolumn{2}{L{\textwidth}}{\descriptionstyle{#5}}
  \end{tabular*}%
}

% Bullet list inside an entry. The class version ends with \vspace{-4mm}, which
% newer array packages (m-cell handling changed 2022/09) honor, making entries
% overlap; older ones (e.g. TeX Live 2022) need it to avoid a gap after entries.
\renewenvironment{cvitems}{%
  \vspace{-4.0mm}
  \begin{justify}
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep]
    \setlength{\parskip}{0pt}
    \renewcommand{\labelitemi}{\bullet}
}{%
  \end{itemize}
  \end{justify}
  \IfPackageAtLeastTF{array}{2022/09/04}{}{\vspace{-4.0mm}}%
}

% Skills with labels flush left; the class version overflows \textwidth

}{%
    \end{tabular*}
  \end{center}
}
